\begin{column}{0.48\textwidth}

  \blocklabel{Motivation}
  \footnotesize

  \begin{itemize}
    \setlength{\itemsep}{0.09cm}
    \setlength{\topsep}{0pt}
    \justifying

    \item \textbf{Gender inequality and poverty:}
    \textbf{\textcolor{boxframe}{85\%}} of single
    parents are women; \textbf{\textcolor{boxframe}{31.4\%}}
    of single-parent households lived in poverty
    in 2022.

    \item \textbf{Unreliable child support:}
    Child support accounts for
    \textbf{\textcolor{boxframe}{18\%}} of
    single-parent families' resources, yet
    \textbf{\textcolor{boxframe}{30--40\%}} of
    payments were fully or partially unpaid.

    \item \textbf{Policy reform:}
    France shifted from ex-post recovery to
    \textbf{automatic public intermediation}
    (March 2022; January 2023).

  \end{itemize}

  \vspace{0.03cm}
  {\tiny\color{textgrey}
  Sources: CNAF (2023); INSEE (2024);
  Ministry of Justice (2022).}

\end{column}

\begin{frame}{Expected Results}

\blocklabel{Hypotheses}
\begin{itemize}
    \item {Take-up of intermediation} should increase sharply: a necessary condition for any other effect.
    \item {Fewer defaults}, but a limited effect: intermediation cannot help when the debtor is insolvent.
    \item {Mothers' labour supply}: ambiguous effect, income effect vs.\ less need to work as a precaution against defaults.
    \item {Living standards}: higher and more stable income, partly offset by lower RSA and prime d'activité.

\end{itemize}
