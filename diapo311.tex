\begin{frame}{Data: Sample Construction and Observation Unit}
\small
\begin{enumerate}\setlength{\itemsep}{0.22cm}
\item \textbf{Start from monthly CAF beneficiary records} in available ALLSTAT-FR6 vintages (target: 2017--2025).
\item \textbf{Identify single-parent family spells} using recorded household/family composition; distinguish entry into single parenthood from first observation in CAF data.
\item \textbf{Recover child support information} and flag observations with usable measures of pension obligations/receipts; document missing and zero values separately.
\item \textbf{Build a longitudinal panel} using stable case/beneficiary identifiers, checking changes of allocataire, CAF transfers and family composition.
\item \textbf{Define policy cohorts only if feasible:} date of establishment of an enforceable child support order, and type of order; otherwise no order-based treatment indicator.
\end{enumerate}
\vspace{0.12cm}
\begin{block}{What does one row represent?}
One CAF family case in month $t$, with observed family composition, child support-related fields, benefit amounts and reported resources.
\end{block}
\smallnote{Selection concern: CAF records do not represent all separated parents; case entry/exit may correlate with outcomes.}
\end{frame}

\begin{frame}{Data: Policy Exposure and Measurement Challenges}
\small
\begin{block}{Institutional variation}
\textbf{March 2022:} automatic intermediation for newly fixed child support in judicial divorce proceedings.\par
\textbf{January 2023:} extension to other relevant enforceable titles. Existing orders are not automatically equivalent to new orders under these rules.
\end{block}
