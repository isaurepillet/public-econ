\begin{frame}{Data: Sample Construction and Observation Unit}

\small
\begin{enumerate}
\setlength{\itemsep}{0.20cm}

\item \textbf{Identify the study population.}
Select single-parent CAF households with dependent children
using \texttt{SITFAM} (family status),
\texttt{PRESCONJ} (partner presence) and
\texttt{NBENLEFA} (dependent children).

\item \textbf{Construct longitudinal observations.}
Use reference dates and anonymised identifiers
(\texttt{DTREFFRE}, \texttt{IDANORES}) to assess
whether households can be followed across periods.
Check changes in beneficiary status and CAF transfers.

\item \textbf{Identify child support arrangements.}
Use \texttt{TPA} (child support indicator),
\texttt{MTPADU} (amount due) and
\texttt{MTALIMFO} (amount received by the household).
Distinguish zero amounts from missing values.

\item \textbf{Characterise family transitions.}
Use \texttt{NBMSIFAM} (months since the last change
in family status) to approximate recent changes,
without interpreting them automatically as separations.


\item \textbf{Assess treatment identification.}
Investigate \texttt{NATJUG} (nature of the judicial decision
concerning a defaulting child support debtor) and the
availability of enforceable-order dates before defining
reform-exposure cohorts.


\end{enumerate}


\begin{block}{Proposed observation unit}
One CAF beneficiary case at each available reference date,
with child support information, family characteristics,
social benefits and reported resources.
\end{block}

\end{frame}

\begin{block}{Measurement issue}
Recorded amounts may reflect declared resources rather
than verified payments. Amounts due and received cannot
yet be used to calculate an arrears rate.
\end{block}

\end{frame}

\begin{frame}{Data: Policy Exposure and Measurement Challenges}
\small
\begin{block}{Institutional variation}
\textbf{March 2022:} automatic intermediation for newly fixed child support in judicial divorce proceedings.\par
\textbf{January 2023:} extension to other relevant enforceable titles. Existing orders are not automatically equivalent to new orders under these rules.
\end{block}
